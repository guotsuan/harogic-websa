cmake_minimum_required(VERSION 3.15)
project(WebSAHarogicShim LANGUAGES CXX)

set(HTRA_SDK_DIR "" CACHE PATH "Directory containing the Windows x64 HTRA headers and import library")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure this adapter for Windows x64")
endif()
foreach(file htra_api.h htra_api_pnm.h htra_api.lib)
    if(NOT EXISTS "${HTRA_SDK_DIR}/${file}")
        message(FATAL_ERROR "HTRA_SDK_DIR is missing ${file}")
    endif()
endforeach()

add_library(websa_san90_sdk_session SHARED san90_sdk.cpp)
target_compile_features(websa_san90_sdk_session PRIVATE cxx_std_17)
target_include_directories(websa_san90_sdk_session PRIVATE "${HTRA_SDK_DIR}")
target_link_libraries(websa_san90_sdk_session PRIVATE "${HTRA_SDK_DIR}/htra_api.lib")
if(MSVC)
    target_compile_options(websa_san90_sdk_session PRIVATE /utf-8)
endif()
install(TARGETS websa_san90_sdk_session RUNTIME DESTINATION .)
